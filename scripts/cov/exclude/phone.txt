# Coverage waivers for the hand-written phone SoC bench (phone_tb_top, scope ^rtl/phone/).
# Scored on the merge of the traffic-model build (TB=phone) and the real-IP
# build (TB=phone_real); see scripts/cov/phone_cov.sh.
# Format: <metric> <regex on "metric hierarchy comment [file:line]"> # reason
# Every waiver must match at least one unhit point (score_cov.py fails otherwise).

# --- Unreachable FSM defaults (Verilator is 2-state; every encoding is a named state)
line u_iommu case rtl/phone/phone_iommu\.sv:(284|355|432)$ # default arm of the walker / read / write FSMs: all encodings are named states
line u_noc case rtl/phone/phone_noc\.sv:(179|292)$ # default arm of the NoC write / read FSMs: all encodings are named states

# --- AxSIZE[2]: the data bus is 32 bits, so AxSIZE <= 2 (4 bytes)
toggle dut(\.u_noc)? (s|m|n|ch)_a[rw]size\[\d+\]\[2\] # 32-bit data bus: AxSIZE <= 2, bit 2 is always 0
toggle u_iommu ([udn]_a[rw]size|[rw]_size)\[2\] # 32-bit data bus: AxSIZE <= 2, bit 2 is always 0

# --- Masters 2..7 sit behind an IOMMU with 3-bit IDs; ID bit 3 marks the
#     IOMMU's own page-table walker, which only reads, and never on the
#     master side of the IOMMU
toggle dut(\.u_noc)? [mn]_(aw|ar|b|r)id\[[2-7]\]\[3\] # IOMMU masters have 3-bit IDs; ID[3] is the walker tag, never driven by the master
toggle u_iommu ([udw]_(aw|ar|b|r)?id|[rw]_id)\[3\] # ID[3] is the walker tag: master-side IDs are 3 bits

# --- The NoC sends ID 0 to the SRAM (one burst at a time)
toggle dut sram_[br]id # SRAM BID/RID echo the tied-0 ID

# --- APB bridge (slave 33) answers OKAY or SLVERR only; the default slave
#     (34) is a constant DECERR responder that never deasserts READY
toggle dut(\.u_noc)? [sp]_[br]resp\[33\]\[0\] # APB bridge answers OKAY/SLVERR only, RESP[0] tied 0
toggle u_noc p_(([br]resp\[34\]\[[01]\])|(a[rw]ready|wready|bvalid|rvalid)\[34\]):1->0 # default slave: constant DECERR, READY tied 1 (only drops at reset, which is never reasserted)

# --- APB window 0x1000_0000..0x1001_FFFF: PADDR[31:17] constant
toggle dut (apbx_|ssp_)?paddr\[(1[7-9]|2\d|3[01])\] # APB window 0x1000_0000..0x1001_FFFF: PADDR[31:17] constant (bit 28 drops only at reset)

# --- IOMMU register blocks are zero-wait: PREADY tied 1
toggle dut (mmu_pready\[\d\]|sl_pready\[(7|8|9|1[0-2])\]):1->0 # IOMMU register blocks are zero-wait, PREADY tied 1
toggle u_iommu pready:1->0 # IOMMU register block is zero-wait, PREADY tied 1

# --- IOMMU internals
toggle u_iommu k_pte_addr\[[01]\] # PTEs are 4-byte words: PTE address bits [1:0] are always 0
toggle u_iommu win_lo\[20\] # win_lo = {1'b0, VA_BASE[31:12]}: bit 20 is a constant 0 pad
toggle u_iommu (n_hits|n_miss|f_cnt)\[(8|9|[12]\d|3[01])\] # 32-bit event counters: bits [7:0] toggle both ways (260+ events per IOMMU); bit 8 1->0 needs 512 and bit 31 needs 2^31 events on one +1 adder

# --- Reset is applied once, at time 0, and never reasserted
toggle (dut|u_noc|u_iommu) (aresetn|rst_n):1->0 # reset is applied once at time 0 (a mid-run reset is covered by the phone_aon unit bench)

# --- Subsystem IP pins and the real-IP wrapper (phone_ss_ips, TB=phone_real).
#     Scope is the SoC glue (Tony, 2026-11-13: "soc glue"; IP pins waived).
#     The pins carry IP function (GPIO/UART/SPI/I2C/TRNG/PWM/IRQ lines) and
#     are closed in open-ip-portfolio's per-IP benches.
toggle dut (gpio|irq)\[\d+\] # IP pin: value set by the GPIO / interrupt sources, closed in the open-ip-portfolio IP benches
toggle dut (i2c_s(cl|da)|spi_(csn|sclk|mosi|miso)|uart_(txd|rxd)|pwm_out|trng_\w+) # IP pin: closed in the open-ip-portfolio IP benches
toggle dut ssp_pslverr:1->0 # subsystem segment PSLVERR: set by the IP's register decode, closed in the IP benches
#     phone_ss_ips is wiring between the IPs' own AXI master / APB / sideband
#     ports and the SoC: every line and branch of it is hit (100%), but which
#     ID, LEN, SIZE, PROT, STRB, BURST and data bits move is decided by each
#     IP's DMA engine, not by the SoC. Those toggles are closed per IP in
#     open-ip-portfolio (PR #33 IPs); the phone_aon_arb ports (g_aon) by the
#     phone_aon unit bench (scripts/cov/exclude/phone_aon.txt).
toggle phone_tb_top\.u_ss  # real-IP wrapper ports: values set by the IPs' DMA engines (closed in the open-ip-portfolio IP benches and the phone_aon unit bench); wrapper lines/branches are 100% hit
